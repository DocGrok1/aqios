import numpy as np
import matplotlib.pyplot as plt

def contact_hamiltonian_flow(x, p, gamma=0.1):
    dxdt = p
    dpdt = -x - gamma * p
    return dxdt, dpdt

def simulate_flow(x0, p0, steps=200, dt=0.05, gamma=0.1):
    xs, ps = [x0], [p0]
    x, p = x0, p0
    for _ in range(steps):
        dxdt, dpdt = contact_hamiltonian_flow(x, p, gamma)
        x += dxdt * dt
        p += dpdt * dt
        xs.append(x)
        ps.append(p)
    return np.array(xs), np.array(ps)
